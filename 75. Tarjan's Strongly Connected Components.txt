75. Tarjan's Strongly Connected Components
import java.util.*;

public class Main {
    static int time = 0;

    static void dfs(int u, List<List<Integer>> graph,
                    int[] disc, int[] low, boolean[] stackMember,
                    Stack<Integer> stack) {

        disc[u] = low[u] = ++time;

        stack.push(u);
        stackMember[u] = true;

        for (int v : graph.get(u)) {
            if (disc[v] == -1) {
                dfs(v, graph, disc, low, stackMember, stack);
                low[u] = Math.min(low[u], low[v]);
            } else if (stackMember[v]) {
                low[u] = Math.min(low[u], disc[v]);
            }
        }

        if (low[u] == disc[u]) {
            int v;

            do {
                v = stack.pop();
                stackMember[v] = false;
                System.out.print(v + " ");
            } while (v != u);

            System.out.println();
        }
    }

    public static void main(String[] args) {
        int n = 5;

        List<List<Integer>> graph = new ArrayList<>();

        for (int i = 0; i < n; i++)
            graph.add(new ArrayList<>());

        graph.get(0).add(2);
        graph.get(2).add(1);
        graph.get(1).add(0);
        graph.get(0).add(3);
        graph.get(3).add(4);

        int[] disc = new int[n];
        int[] low = new int[n];

        Arrays.fill(disc, -1);

        boolean[] stackMember = new boolean[n];
        Stack<Integer> stack = new Stack<>();

        for (int i = 0; i < n; i++) {
            if (disc[i] == -1)
                dfs(i, graph, disc, low, stackMember, stack);
        }
    }
}
